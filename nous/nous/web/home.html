接入网关

# 把原来的模型地址换成网关地址，其余代码不变
curl http://HOST:8080/v1/chat/completions \
  -H "Authorization: Bearer $NOUS_API_KEY" -H "Content-Type: application/json" \
  -d '{"messages":[{"role":"user","content":"你好"}],"stream":true}'
# 每个流式片段多一个 x_risk 字段：回复前 {"phase":"pre","C1_pre","C2"}，逐字 {"C1_mean","alarm":{"C1_watch","C1"}}

接入陪伴设备

# WebSocket，一台设备一个密钥
ws://HOST:8080/v1/companion/ws?key=$DEVICE_KEY
→ {"type":"chat","text":"我今天考了第一名！"}
← {"type":"expression","phase":"pre","expr":{"happy":0.71,...}}   # 第一个字之前
← {"type":"sentence","text":"太棒了！","expr":{...}}                # 一句一句放行给语音合成
← {"type":"done","reply":"..."}
